```go
func TestMVStorePutAndGet(t *testing.T) {
    store := NewMemoryStore()

    // Given
    key := "test-key"
    value := []byte("test-value")

    // When
    err := store.Put(key, value)
    assert.NoError(t, err)

    // Then
    result, err := store.Get(key)
    assert.NoError(t, err)
    assert.Equal(t, value, result)

func TestMVStoreGetVersion(t *testing.T) {
    store := NewMemoryStore()

    // Given
    key := "test-key"
    v1 := []byte("value-v1")
    v2 := []byte("value-v2")

    store.Put(key, v1)  // version 1
    store.Put(key, v2)  // version 2

    // When
    result, err := store.GetVersion(key, 1)

    // Then
    assert.NoError(t, err)
    assert.Equal(t, v1, result)
}

func TestMVStoreDelete(t *testing.T) {
```

#### WAL 测试

```go
func TestWALAppendAndRecover(t *testing.T) {
    file := "test.wal"
    defer os.Remove(file)

    wal, err := OpenWAL(file)
    require.NoError(t, err)
    defer wal.Close()

    // Given
    entries := []*WALEntry{
        {Sequence: 1, Type: WALTypePut, Key: "key1", Value: []byte("value1")},
        {Sequence: 2, Type: WALTypePut, Key: "key2", Value: []byte("value2")},
    }

    // When
    for _, entry := range entries {
        err := wal.Append(entry)
        assert.NoError(t, err)
    }

    err = wal.Sync()
    assert.NoError(t, err)

    // Recover
    wal2, err := OpenWAL(file)
    require.NoError(t, err)
    defer wal2.Close()

    recovered, err := wal2.Recover()
    assert.NoError(t, err)

    // Then
    assert.Equal(t, len(entries), len(recovered))
}
```

```go
func TestGossipSync(t *testing.T) {
    // Given
    node1 := NewTestNode(t, "node-1")
    node2 := NewTestNode(t, "node-2")

    node1.Put("key1", []byte("value1"))  // version 1

    // When
    node2.GossipTo(node1.Addr())

    // Then
    value, err := node2.Get("key1")
    assert.NoError(t, err)
    assert.Equal(t, []byte("value1"), value)
}

func TestGossipConvergence(t *testing.T) {
    // Given
    nodes := []*TestNode{
        NewTestNode(t, "node-1"),
        NewTestNode(t, "node-2"),
        NewTestNode(t, "node-3"),
    }

    nodes[0].Put("key1", []byte("value1"))

    // When
    for _, node := range nodes {
        for _, other := range nodes {
            if node != other {
                node.GossipTo(other.Addr())
            }
        }
    }

    // Then
    for _, node := range nodes {
        value, err := node.Get("key1")
        assert.NoError(t, err)
        assert.Equal(t, []byte("value1"), value)
    }
}
```

```go
func TestQuorumSuccess(t *testing.T) {
    // Given
    nodes := []*TestNode{
        NewTestNode(t, "node-1"),
        NewTestNode(t, "node-2"),
        NewTestNode(t, "node-3"),
    }

    change := &MetadataChange{ID: "change-1", Type: ChangeTypePut}

    // When
    err := nodes[0].ProposeChange(change, nodes[1:])
    assert.NoError(t, err)

    // Then
    for _, node := range nodes {
        applied, err := node.IsChangeApplied(change.ID)
        assert.NoError(t, err)
        assert.True(t, applied)
    }
}

func TestQuorumTimeout(t *testing.T) {
    // Given
    nodes := []*TestNode{
        NewTestNode(t, "node-1"),
        NewTestNode(t, "node-2"),
        NewTestNode(t, "node-3"),
    }

    // node-2 和 node-3 会拒绝请求
    change := &MetadataChange{ID: "change-1", Type: ChangeTypePut}
    nodes[1].RejectAll()
    nodes[2].RejectAll()

    // When
    err := nodes[0].ProposeChange(change, nodes[1:])

    // Then
    assert.Error(t, err)
    assert.Equal(t, ErrQuorumTimeout, err)
}
```

```go
    assert.Len(t, children, 3)

func TestFailureDetection(t *testing.T) {
    // Given
    detector := NewFailureDetector()
    detector.AddNode("node-1", "127.0.0.1:9211")
    detector.SetSuspectTimeout(1 * time.Second)

    // When: 模拟节点故障
    detector.StopHeartbeat("node-1")
    time.Sleep(2 * time.Second)

    // Then
    status := detector.GetStatus("node-1")
    assert.Equal(t, StatusSuspect, status)

    // 再等待更长时间
    time.Sleep(2 * time.Second)
    status = detector.GetStatus("node-1")
    assert.Equal(t, StatusDead, status)
}
```

### 3.3 故障注入测试

```go
func TestNodeFailure(t *testing.T) {
    // Given: 3 节点集群
    cluster := NewTestCluster(t, 3)
    defer cluster.Stop()

    cluster.Put("key1", []byte("value1"))

    // When: node-2 故障
    cluster.KillNode("node-2")

    // Then: 剩余节点仍然可用
    value, err := cluster.Node("node-1").Get("key1")
    assert.NoError(t, err)
    assert.Equal(t, []byte("value1"), value)

    // When: node-2 恢复
    cluster.RestartNode("node-2")

    // Then: 自动恢复数据
    value, err = cluster.Node("node-2").Get("key1")
    assert.NoError(t, err)
    assert.Equal(t, []byte("value1"), value)
}
```

## 6. 持续集成

### 6.1 CI 流程

```yaml
# .github/workflows/test.yml
name: Test

on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2

      - name: Set up Go
        uses: actions/setup-go@v2
        with:
          go-version: 1.21

      - name: Run tests
        run: |
          go test -v -race -coverprofile=coverage.out ./...

      - name: Upload coverage
        uses: codecov/codecov-action@v2
        with:
          files: ./coverage.out

      - name: Run benchmarks
        run: |
          go test -bench=. -benchmem ./...
```
