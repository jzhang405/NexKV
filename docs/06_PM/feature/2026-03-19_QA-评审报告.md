#### R1 详情：Leaf Lock 死锁风险

**场景**：
```go
// 线程 1: Set(key1) → Lock LeafA → 分裂 → 尝试 Lock Parent
// 线程 2: Set(key2) → Lock Parent → 尝试 Lock LeafA
// 结果：死锁（循环等待）
```

**文档当前状态**：
- 第 276 行："严格加锁顺序：总是先 Leaf 后 Root"
- ❌ **问题**：分裂时需要向上传播，会打破这个顺序
- ❌ **问题**：没有说明如何检测死锁
- ❌ **问题**：没有超时机制的具体实现

**必需措施**：
1. 添加死锁检测测试（详见 3.1.4）
2. 实现锁超时机制（建议 5ms 超时）
3. 分裂期间使用 tryLock + 重试，避免阻塞

func (b *BTree) GetLock(pageID uint64) *PageLock {

    lock := NewPageLock()
    b.pageLocks.Add(pageID, lock)
    return lock

#### ❌ 测试 1：死锁检测测试（**P0**）

**目的**：验证分裂期间的加锁顺序不会导致死锁

**测试场景**：
```go
func TestBTree_LeafLock_Deadlock_DuringSplit(t *testing.T) {
    tree := setupTestTree(t)

    // 场景 1：并发写入触发分裂
    const goroutines = 100
    const writesPerGoroutine = 1000

    var wg sync.WaitGroup
    for i := 0; i < goroutines; i++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            for j := 0; j < writesPerGoroutine; j++ {
                key := fmt.Sprintf("key-%d-%d", id, j)
                value := []byte(fmt.Sprintf("value-%d", j))

                // 设置超时，防止死锁
                ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
                defer cancel()

                err := tree.Set(ctx, key, value)
                require.NoError(t, err) // 超时会返回错误
            }
        }(i)
    }

    done := make(chan struct{})
    go func() {
        wg.Wait()
        close(done)
    }()

    select {
    case <-done:
        // 成功完成，无死锁
    case <-time.After(30 * time.Second):
        t.Fatal("deadlock detected: test did not complete in 30s")
    }
}
```

**预期执行时间**：~30 秒

func TestBTree_LeafLock_ConcurrentSplit_Correctness(t *testing.T) {
    tree := setupTestTree(t)

    // 使用小的页大小，频繁触发分裂
    tree.config.MaxLeafKeys = 50

    // 并发写入
    const goroutines = 50
    const writesPerGoroutine = 500

    results := make(chan error, goroutines)
    var wg sync.WaitGroup

    for i := 0; i < goroutines; i++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            for j := 0; j < writesPerGoroutine; j++ {
                key := fmt.Sprintf("key-%d", j%250) // 重复键，触发更新
                value := []byte(fmt.Sprintf("value-%d-from-goroutine-%d", j, id))

                err := tree.Set(context.Background(), key, value)
                if err != nil {
                    results <- fmt.Errorf("goroutine %d: %w", id, err)
                    return
                }
            }
        }(i)
    }

    wg.Wait()
    close(results)

    // 检查错误
    for err := range results {

    }

    // 验证数据完整性
    for j := 0; j < 250; j++ {
        key := fmt.Sprintf("key-%d", j)
        value, err := tree.Get(context.Background(), []byte(key))
        require.NoError(t, err)
        require.NotEmpty(t, value, "key %s should have a value", key)
    }
}

func BenchmarkBTree_LeafLock_Contention(b *testing.B) {
    tree := setupBenchmarkTree(b)

    scenarios := []struct {
        name         string
        hotKeyRatio  float64 // 热键比例
        goroutines   int
    }{
        {"AllHotKeys", 1.0, 8},    // 100% 热键（最大竞争）
        {"MostlyHot", 0.8, 8},     // 80% 热键
        {"Balanced", 0.5, 8},      // 50% 热键
        {"MostlyCold", 0.2, 8},    // 20% 热键
        {"AllCold", 0.0, 8},       // 0% 热键（无竞争）
    }

    for _, scenario := range scenarios {
        b.Run(scenario.name, func(b *testing.B) {
            b.RunParallel(func(pb *testing.PB) {
                // 根据热键比例选择键
                hotKeys := int(100 * scenario.hotKeyRatio)
                coldKeys := 100 - hotKeys

                for pb.Next() {
                    var key string
                    if rand.Float64() < scenario.hotKeyRatio {
                        key = fmt.Sprintf("hot-key-%d", rand.Intn(hotKeys))
                    } else {
                        key = fmt.Sprintf("cold-key-%d", rand.Intn(coldKeys))
                    }

                    value := []byte(fmt.Sprintf("value-%d", rand.Intn(1000)))
                    _ = tree.Set(context.Background(), []byte(key), value)
                }
            })
        })
    }
}

func (b *BTree) Set(ctx context.Context, key, value []byte) error {

func TestBTree_LeafLock_Deadlock_Template(t *testing.T) {
    tree := setupTestTree(t)

    // 配置：小的页大小，频繁触发分裂
    tree.config.MaxLeafKeys = 50

    // 并发写入
    const goroutines = 100
    const writesPerGoroutine = 1000

    done := make(chan struct{})
    go func() {
        var wg sync.WaitGroup
        for i := 0; i < goroutines; i++ {
            wg.Add(1)
            go func(id int) {
                defer wg.Done()
                for j := 0; j < writesPerGoroutine; j++ {
                    key := fmt.Sprintf("key-%d-%d", id, j)
                    value := []byte(fmt.Sprintf("value-%d", j))
                    tree.Set(context.Background(), []byte(key), value)
                }
            }(i)
        }
        wg.Wait()
        close(done)
    }()

    select {
    case <-done:
        t.Log("No deadlock detected")
    case <-time.After(30 * time.Second):
        t.Fatal("Deadlock detected")
    }
}
