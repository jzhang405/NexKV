#### 2. Scan/Iterator Delta Chain 支持（2 小时）

**文件**：`iterator.go`（修改）

**修改内容**：
- `Next()` 方法：添加 Delta Chain 遍历
- 遍历顺序：**Mini-Page 优先**，然后 Delta Chain
- 使用 `deletedInDeltaChain` 和 `keyInMiniPage` 辅助函数

**实现方案**：
```go
// 辅助函数：检查键是否在 Delta Chain 中被删除
func deletedInDeltaChain(key []byte, deltaChain *DeltaChain) bool {
    for _, delta := range deltaChain.deltas {
        if delta.op == DeltaOpDelete && bytes.Equal(delta.key, key) {
            return true
        }
    }
    return false
}

// 辅助函数：检查键是否在 MiniPage 中
func keyInMiniPage(key []byte, miniPage *MiniPage) bool {
    _, found := miniPage.findSlot(key)
    return found >= 0
}

// Next 方法的实现
func (it *ScanIterator) Next() (valid bool, key []byte, value []byte, err error) {
    // ... 获取 leafNode ...

    // 1. 先遍历 MiniPage
    for top.index < len(leafNode.miniPage.slots) {
        slot := leafNode.miniPage.slots[top.index]

        // 检查范围
        if it.start != nil && compareKeys(slot.key, it.start) < 0 {
            top.index++
            continue
        }
        if it.end != nil && compareKeys(slot.key, it.end) >= 0 {
            break
        }

        // 检查是否在 Delta Chain 中被删除
        if !deletedInDeltaChain(slot.key, leafNode.deltaChain) {
            // 返回 Mini-Page 中的键值对
            top.index++
            it.current = slot.key
            return true, slot.key, slot.value, nil
        }

        top.index++
    }

    // 2. 再遍历 Delta Chain（排除 MiniPage 中的键）
    for _, delta := range leafNode.deltaChain.deltas {
        if delta.op == DeltaOpDelete {
            continue
        }
        if it.start != nil && compareKeys(delta.key, it.start) < 0 {
            continue
        }
        if it.end != nil && compareKeys(delta.key, it.end) >= 0 {
            continue
        }
        // 检查键是否已在 MiniPage 中返回过
        if !keyInMiniPage(delta.key, leafNode.miniPage) {
            return true, delta.key, delta.value, nil
        }
    }

    // 没有更多数据
    return false, nil, nil, nil
}
```

```go
const MaxSplitDepth = 10 // 最大 10 层

func (t *BfTree) insertSplitIntoParent(parentPageID, leftPageID, rightPageID uint64, splitKey []byte) error {
    return t.insertSplitWithDepth(parentPageID, leftPageID, rightPageID, splitKey, 0)
}

func (t *BfTree) insertSplitWithDepth(parentID, leftPageID, rightPageID uint64, splitKey []byte, depth int) error {
    // 递归深度检查
    if depth > MaxSplitDepth {
        return ErrTreeTooDeep
    }

    // 获取父节点
    parentNode, err := t.pageStore.getInner(parentPageID)
    if err != nil {
        return err
    }

    // 尝试插入分隔键和子节点
    if !parentNode.IsFull() {
        // 直接插入
        return parentNode.InsertChild(...)
    }

    // 父节点也满了，需要分裂
    newLeft, newRight, newSplitKey, oldParentID, err := t.splitInnerNode(parentPageID)
    if err != nil {
        return err
    }

    // 释放旧父节点
    _ = t.pageTable.Free(oldParentID)

    // 递归向上
    grandParentID := t.findParent(parentPageID)
    if grandParentID == 0 || grandParentID == parentPageID {
        // 到达根节点，创建新根
        return t.createNewRoot(newLeft, newRight, newSplitKey)
    }

    return t.insertSplitWithDepth(grandParentID, newLeft, newRight, newSplitKey, depth+1)
}
```

### 节点合并算法（简化版）

```go
func (t *BfTree) tryMergeAfterDelete(leafPageID uint64) error {
    // 1. 获取当前节点
    node, err := t.pageStore.getLeaf(leafPageID)
    if err != nil {
        return err
    }

    // 2. 获取左右兄弟节点
    leftSibling, rightSibling, err := t.getSiblings(leafPageID)
    if err != nil {
        return nil // 无兄弟节点，不合并
    }

    // 3. 检查是否可以合并
    totalSize := node.size() + leftSibling.size() + rightSibling.size()
    if totalSize > t.config.PageSize {
        return nil // 太大，不合并
    }

    // 4. 执行合并
    mergedNode := mergeLeafNodes(leftSibling, node, rightSibling)

    // 5. 更新父节点
    return t.updateParentAfterMerge(...)
}
```

### Delta Chain 查重辅助函数

```go
// deletedInDeltaChain 检查键是否在 Delta Chain 中被删除
func deletedInDeltaChain(key []byte, deltaChain *DeltaChain) bool {
    // 从后向前遍历（最新的优先）
    for i := len(deltaChain.deltas) - 1; i >= 0; i-- {
        delta := deltaChain.deltas[i]
        if delta.op == DeltaOpDelete && bytes.Equal(delta.key, key) {
            return true
        }
    }
    return false
}

// keyInMiniPage 检查键是否在 MiniPage 中
func keyInMiniPage(key []byte, miniPage *MiniPage) bool {
    index, found := miniPage.findSlot(key)
    return found && index < len(miniPage.slots)
}
```
